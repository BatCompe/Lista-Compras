# Lê todos os dias os preços dos produtos de produtos.json (Pingo Doce e Continente) e grava precos.json.
# Também se pode correr à mão: separador Actions › "Atualizar preços" › Run workflow.
name: Atualizar preços

on:
  schedule:
    - cron: '17 5 * * *'      # todos os dias às 5:17 (UTC)
  workflow_dispatch:
  push:
    paths: ['produtos.json']  # quando se carrega um produtos.json novo

permissions:
  contents: write

concurrency:
  group: precos
  cancel-in-progress: false

jobs:
  precos:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Instalar
        run: |
          pip install requests beautifulsoup4 playwright
          python -m playwright install --with-deps chromium
      - name: Ler preços
        run: python scripts/precos.py
      - name: Guardar precos.json
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add precos.json
          git diff --cached --quiet || (git commit -m "Preços atualizados $(date -u +%Y-%m-%d)" && git push)
